import { describe, it } from "vitest";
import { writeFileSync, mkdirSync } from "node:fs";
import { layoutElkGraph } from "@/features/canvas/layout/layoutEngine";
import { irToLayoutGraph } from "@/features/llm/ir/ir-to-layout-graph";
import {
  measureReadability,
  measurePolylines,
  type ReadabilityReport,
} from "@/features/canvas/layout/layoutReadability";
import { readElkHandleOrder } from "@/features/canvas/layout/elkHandleOrder";
import {
  buildRenderedPolylines,
  measureRenderedReadability,
} from "@/features/canvas/layout/renderedEdgePath";
import {
  handPlacedDiagram,
  handPlacedLabels,
  handPlacedParents,
} from "@/features/canvas/layout/hand-placed-diagram";
import { labelsOf, REFERENCE_DIAGRAMS } from "@/features/canvas/layout/reference-diagrams";
import type { DiagramIR } from "@/features/llm/ir/ir.types";

const OUT = process.env.PROBE_OUT ?? "/tmp/probe";
mkdirSync(OUT, { recursive: true });
const RUNS = Number(process.env.PROBE_RUNS ?? "10");

function dump(name: string, data: unknown) {
  writeFileSync(`${OUT}/${name}.json`, JSON.stringify(data, null, 2));
}

async function measureOne(ir: DiagramIR) {
  const graph = await layoutElkGraph(irToLayoutGraph(ir));
  const labels = labelsOf(ir);
  const elk = measureReadability(graph, { labels });
  const rendered = measureRenderedReadability(graph, ir.edges, {
    labels,
    handleOrder: readElkHandleOrder(graph),
  });
  const roundRobin = measureRenderedReadability(graph, ir.edges, { labels });
  return {
    elkCrossings: elk.edgeCrossings,
    elkOverNode: elk.edgeNodeOverlaps,
    placement: elk.placementCrossings,
    renderedCrossings: rendered.edgeCrossings,
    renderedOverNode: rendered.edgeNodeOverlaps,
    renderedLabels: rendered.labelOverlaps,
    roundRobinCrossings: roundRobin.edgeCrossings,
    nodeCount: elk.nodeCount,
    edgeCount: elk.edgeCount,
    width: elk.width,
    height: elk.height,
  };
}

function measureHandPlaced(): ReadabilityReport {
  const d = handPlacedDiagram();
  return measurePolylines(
    {
      boxes: d.boxes,
      parentOf: handPlacedParents(d),
      edges: buildRenderedPolylines(
        d.boxes,
        d.edges.map((e) => ({ id: e.id, sourceId: e.sourceId, targetId: e.targetId })),
      ),
      rootId: d.rootId,
      width: d.width,
      height: d.height,
    },
    { labels: handPlacedLabels(d) },
  );
}

describe("probe", () => {
  it(`A: repeats the five reference measurements ${RUNS}x`, async () => {
    const runs: Record<string, unknown>[] = [];
    for (let i = 0; i < RUNS; i++) {
      const row: Record<string, unknown> = {};
      for (const { name, ir } of REFERENCE_DIAGRAMS) {
        row[name] = await measureOne(ir);
      }
      const hp = measureHandPlaced();
      row["Hand-placed"] = {
        renderedCrossings: hp.edgeCrossings,
        renderedOverNode: hp.edgeNodeOverlaps,
        renderedLabels: hp.labelOverlaps,
        nodeCount: hp.nodeCount,
        edgeCount: hp.edgeCount,
      };
      runs.push(row);
    }
    dump("A-repeat", { runs: RUNS, data: runs });
  }, 600000);

  it("B: proves each ELK option is read, by perturbing it", async () => {
    const ir = REFERENCE_DIAGRAMS[1]!.ir; // AWS ECS Fargate (nested)
    const graph = irToLayoutGraph(ir);
    const fingerprint = (g: Awaited<ReturnType<typeof layoutElkGraph>>) => {
      const boxes: string[] = [];
      const walk = (n: typeof g) => {
        boxes.push(`${n.id}:${n.x ?? 0},${n.y ?? 0},${n.width ?? 0},${n.height ?? 0}`);
        for (const c of n.children ?? []) walk(c as typeof g);
      };
      walk(g);
      return boxes.sort().join("|");
    };

    const base = fingerprint(await layoutElkGraph(graph));

    // Each probe: a value that MUST change the result if ELK reads the key,
    // plus a deliberately misspelled key that MUST NOT change it (control).
    const probes: Record<string, string> = {
      "elk.algorithm": "force",
      "elk.direction": "DOWN",
      "elk.edgeRouting": "POLYLINE",
      "elk.layered.spacing.nodeNodeBetweenLayers": "600",
      "elk.spacing.nodeNode": "400",
      "elk.layered.nodePlacement.strategy": "SIMPLE",
      "elk.padding": "[top=300,left=300,bottom=300,right=300]",
      "elk.hierarchyHandling": "SEPARATE_CHILDREN",
    };
    const results: Record<string, { changed: boolean }> = {};
    for (const [key, value] of Object.entries(probes)) {
      const f = fingerprint(await layoutElkGraph(graph, { [key]: value }));
      results[key] = { changed: f !== base };
    }

    // Controls: keys ELK cannot know. If these "change" the layout the whole
    // experiment is invalid.
    for (const [key, value] of [
      ["elk.paddingg", "[top=300,left=300,bottom=300,right=300]"],
      ["elk.padding.WRONG", "[top=300,left=300,bottom=300,right=300]"],
      ["elk.spacing.nodeNodeTypo", "400"],
      ["elk.direction.LEFT_TO_RIGHT", "LEFT_TO_RIGHT"],
    ] as const) {
      const f = fingerprint(await layoutElkGraph(graph, { [key]: value }));
      results[`CONTROL ${key}`] = { changed: f !== base };
    }

    // The specific claim in the layoutEngine comment: `:` separators in padding
    // are silently ignored.
    const colon = fingerprint(
      await layoutElkGraph(graph, { "elk.padding": "[top:300,left:300,bottom:300,right:300]" }),
    );
    results["CLAIM elk.padding with ':' separators"] = { changed: colon !== base };

    // The other claim: LEFT_TO_RIGHT is not an ELK token for elk.direction.
    const ltr = fingerprint(await layoutElkGraph(graph, { "elk.direction": "LEFT_TO_RIGHT" }));
    const right = fingerprint(await layoutElkGraph(graph, { "elk.direction": "RIGHT" }));
    results["CLAIM elk.direction=LEFT_TO_RIGHT vs base"] = { changed: ltr !== base };
    results["CLAIM elk.direction=RIGHT equals base"] = { changed: right !== base };

    dump("B-elk-options", results);
  }, 600000);

  it("C: measures id sensitivity by relabelling ids structure-identically", async () => {
    const rows: Record<string, unknown> = {};
    for (const { name, ir } of REFERENCE_DIAGRAMS) {
      const crossings: number[] = [];
      for (let seed = 0; seed < 12; seed++) {
        // Structure-preserving relabel: every id replaced by a fresh token in a
        // different collation order. Nothing about the graph changes but names.
        const map = new Map<string, string>();
        ir.nodes.forEach((n, i) => map.set(n.id, `n-${((i * 7 + seed * 13) % 97) + 100}-${seed}`));
        const relabelled: DiagramIR = {
          type: ir.type,
          nodes: ir.nodes.map((n) => ({
            ...n,
            id: map.get(n.id)!,
            parentId: n.parentId === null ? null : (map.get(n.parentId) ?? null),
          })),
          edges: ir.edges.map((e, i) => ({
            ...e,
            id: `e-${((i * 11 + seed * 5) % 97) + 100}-${seed}`,
            sourceId: map.get(e.sourceId)!,
            targetId: map.get(e.targetId)!,
          })),
        };
        const graph = await layoutElkGraph(irToLayoutGraph(relabelled));
        crossings.push(
          measureRenderedReadability(graph, relabelled.edges, {
            handleOrder: readElkHandleOrder(graph),
          }).edgeCrossings,
        );
      }
      rows[name] = {
        crossings,
        min: Math.min(...crossings),
        max: Math.max(...crossings),
        spread: Math.max(...crossings) - Math.min(...crossings),
      };
    }
    dump("C-id-sensitivity", rows);
  }, 600000);
});
