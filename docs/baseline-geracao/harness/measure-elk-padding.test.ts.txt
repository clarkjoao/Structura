import { describe, it } from "vitest";
import { writeFileSync, mkdirSync } from "node:fs";
import { layoutElkGraph } from "@/features/canvas/layout/layoutEngine";
import { irToLayoutGraph } from "@/features/llm/ir/ir-to-layout-graph";
import { REFERENCE_DIAGRAMS } from "@/features/canvas/layout/reference-diagrams";

const OUT = process.env.PROBE_OUT ?? "/tmp/probe";
mkdirSync(OUT, { recursive: true });

describe("probe2", () => {
  it("D: disambiguates the elk.padding separator claim", async () => {
    const ir = REFERENCE_DIAGRAMS[1]!.ir;
    const graph = irToLayoutGraph(ir);

    // The padded container's own size is the direct read-out of padding:
    // a compound node is sized to fit children + padding.
    async function paddedBoxes(padding: string | null) {
      const g = await layoutElkGraph(
        graph,
        padding === null ? {} : { "elk.padding": padding },
      );
      const sizes: Record<string, string> = {};
      const walk = (n: typeof g) => {
        if ((n.children ?? []).length > 0) {
          sizes[n.id] = `${Math.round(n.width ?? 0)}x${Math.round(n.height ?? 0)}`;
        }
        for (const c of n.children ?? []) walk(c as typeof g);
      };
      walk(g);
      return sizes;
    }

    const result = {
      "shipped [top=40,...] (= separators)": await paddedBoxes("[top=40,left=40,bottom=40,right=40]"),
      "explicit 12 (ELK default)": await paddedBoxes("[top=12,left=12,bottom=12,right=12]"),
      "explicit 300": await paddedBoxes("[top=300,left=300,bottom=300,right=300]"),
      "colon separators [top:300,...]": await paddedBoxes("[top:300,left:300,bottom:300,right:300]"),
      "colon separators [top:40,...]": await paddedBoxes("[top:40,left:40,bottom:40,right:40]"),
      "garbage value 'nonsense'": await paddedBoxes("nonsense"),
    };
    writeFileSync(`${OUT}/D-padding.json`, JSON.stringify(result, null, 2));
  }, 600000);
});
